請打開您的 Messenger.html，按 Cmd + F 或 Ctrl + F 搜尋 async function renderChatList(snapshot)，並將整段函式完全覆蓋為以下程式碼（我已經在裡面加入了 [SYS_BLOCK] 的攔截引擎）：
```html
async function renderChatList(snapshot) {
        const listElement = document.getElementById('chatList');
        if (snapshot.empty) { listElement.innerHTML = '<p style="text-align:center; color:#999; margin-top:20px;">暫無對話</p>'; removeInitLoader(); return; }

        const currentUser = auth.currentUser;
        let docs = snapshot.docs;
        
        const pinnedChats = [], normalChats = [];
        docs.forEach(doc => {
            const data = doc.data();
            if (data.isGroup && data.groupName === "立法院 Legislature") pinnedChats.push(doc);
            else normalChats.push(doc);
        });
        const sortedDocs = [...pinnedChats, ...normalChats];

        const htmlPromises = sortedDocs.map(async (doc) => {
            const chat = doc.data(), chatId = doc.id;
            let title = "", avatar = "", avatarStyle = "";

            let targetTheme = 'default';
            if (chat.isGroup) {
                title = chat.groupName || "未命名群組";
                avatar = chat.groupAvatar || `https://ui-avatars.com/api/?name=${encodeURIComponent(title)}&background=003153&color=fff`;
            } else {
                const targetUid = chat.members.find(uid => uid !== window.activeAppUid);
                
                if (!window.actListenersAttached) window.actListenersAttached = {};
                if (targetUid && !window.actListenersAttached[targetUid]) {
                    window.actListenersAttached[targetUid] = true;
                    db.collection("act").doc(targetUid).onSnapshot(userDoc => {
                        if (userDoc.exists) {
                            const data = userDoc.data();
                            userCache[targetUid] = data;
                            window.chatMembersCache[targetUid] = data; 
                            
                            if (window.lastChatsSnapshot) {
                                clearTimeout(window.actUpdateTimer);
                                window.actUpdateTimer = setTimeout(() => {
                                    renderChatList(window.lastChatsSnapshot);
                                }, 300);
                            }
                            if (currentChatId) {
                                clearTimeout(window.chatRoomUpdateTimer);
                                window.chatRoomUpdateTimer = setTimeout(() => {
                                    renderChatUI();
                                    renderTypingUI();
                                }, 300);
                            }
                        }
                    });
                    
                    if (!userCache[targetUid]) {
                        const userDoc = await db.collection("act").doc(targetUid).get();
                        if (userDoc.exists) userCache[targetUid] = userDoc.data();
                    }
                }

                const targetData = userCache[targetUid] || { displayName: "未知成員", photoURL: "https://via.placeholder.com/45" };
                title = targetData.displayName; 
                avatar = targetData.photoURL;
                targetTheme = targetData.userTheme || 'default'; 
                avatarStyle = "width: 60px; height: 60px; margin-right: 10px;"; 
            }

            let displayLastMsg = '尚無訊息';
            let msgTime = chat.updatedAt ? chat.updatedAt.toDate() : new Date();
            let latestVisibleSenderId = chat.lastSenderId;

            try {
                const lastMsgsSnap = await db.collection("chats").doc(chatId).collection("messages")
                    .orderBy("timestamp", "desc").limit(20).get();
                
                for (let mDoc of lastMsgsSnap.docs) {
                    const mData = mDoc.data();
                    if (!mData.hiddenBy || !mData.hiddenBy.includes(currentUser.uid)) {
                        let rawText = mData.text || (mData.fileType === 'video' ? '[影片]' : '[圖片]');
                        if (mData.fileType === 'sticker') rawText = '[貼圖]';
                        else if (mData.fileType === 'audio') rawText = '🎤 [語音訊息]';
                        
                        if (rawText.includes('"type":"event_share"')) {
                            try { const parsed = JSON.parse(rawText); rawText = `分享了活動：${parsed.title}`; } catch(e) {}
                        } else if (rawText.includes('"type":"group_buy"')) {
                            try { const parsed = JSON.parse(rawText); rawText = `發起了團購：${parsed.title}`; } catch(e) {}
                        } else if (rawText.includes('"type":"confidential"')) {
                            try { const parsed = JSON.parse(rawText); rawText = `傳送了機密文件：${parsed.title}`; } catch(e) {}
                        } else if (rawText.includes('"type":"poll"')) {
                            try { const parsed = JSON.parse(rawText); rawText = `發起了投票：${parsed.question}`; } catch(e) {}
                        } 
                        // 👉 關鍵新增：攔截系統封鎖代碼，轉換為友善提示
                        else if (rawText.startsWith('[SYS_BLOCK]')) {
                            rawText = "🚫 對話狀態已變更";
                        }
                        
                        displayLastMsg = rawText;
                        if (mData.timestamp) msgTime = mData.timestamp.toDate();
                        latestVisibleSenderId = mData.senderId;
                        break;
                    }
                }
            } catch (e) {
                displayLastMsg = chat.lastMessage || '尚無訊息';
                // 防呆攔截
                if (displayLastMsg.startsWith('[SYS_BLOCK]')) displayLastMsg = "🚫 對話狀態已變更";
            }

            const myLastRead = chat.lastReadAt?.[currentUser.uid] ? chat.lastReadAt[currentUser.uid].toDate() : new Date(0);
            const isUnread = latestVisibleSenderId !== currentUser.uid && msgTime > myLastRead;
            const unreadClass = isUnread ? "unread" : "";
            
            const pinIcon = (chat.isGroup && chat.groupName === "立法院 Legislature") ? '<i class="fas fa-thumbtack" style="color:var(--ly-gold); position:absolute; top:10px; left:12px; transform:rotate(-45deg); font-size:14px;"></i>' : '';

            let timeStr = "";
            if (msgTime) {
                const now = new Date();
                if (msgTime.toDateString() === now.toDateString()) timeStr = msgTime.toLocaleTimeString('zh-TW', {hour: '2-digit', minute:'2-digit', hour12: false});
                else timeStr = `${msgTime.getFullYear()}/${(msgTime.getMonth() + 1).toString().padStart(2,'0')}/${msgTime.getDate().toString().padStart(2,'0')}`;
            }

            let avatarHtml = `<img class="user-avatar" src="${avatar}" style="${avatarStyle}">`;
            if (!chat.isGroup && targetTheme && targetTheme !== 'default') {
                const themeClass = `theme-${targetTheme}`;
                avatarHtml = `<div class="chat-avatar-wrapper ${themeClass}" style="${avatarStyle}">
                                  <img class="user-avatar" src="${avatar}" style="width: 100%; height: 100%; border: 1.5px solid #fff;">`;
                if (targetTheme === 'princess') {
                    avatarHtml += `<div class="chat-theme-crown"><i class="fas fa-crown"></i></div><div class="chat-theme-sakura sakura-1"></div><div class="chat-theme-sakura sakura-2"></div>`;
                } else if (targetTheme === 'tech') {
                    avatarHtml += `<div class="chat-theme-tech-ring"></div><div class="chat-theme-tech-ring inner" style="inset:-6px; animation-duration: 7s; animation-direction: reverse;"></div>`;
                } else if (targetTheme === 'chuuni') {
                    avatarHtml += `<div class="chat-theme-chuuni-rune"></div>`;
                } 
                else if (targetTheme.startsWith('style')) {
                    avatarHtml += `</div>`;
                    avatarHtml = `<div class="chat-avatar-wrapper theme-${targetTheme}">${avatarHtml}</div>`;
                } else {
                    avatarHtml += `</div>`;
                }
            }

            return `
                <div class="chat-item ${unreadClass}" onclick="openChatRoom('${chatId}')">
                    ${pinIcon} ${avatarHtml}
                    <div class="chat-info">
                        <div style="display:flex; justify-content:space-between; align-items:center;">
                            <h4>${title}</h4> <span class="chat-time">${timeStr}</span>
                        </div>
                        <p>${displayLastMsg}</p>
                    </div>
                </div>`;
        });

        const html = await Promise.all(htmlPromises);
        listElement.innerHTML = html.join("");
        removeInitLoader();
    }
    ```
